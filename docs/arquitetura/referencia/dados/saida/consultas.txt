-- radar: casas ativas da microarea, com as confirmacoes
-- @microarea='3'
SELECT c.casa_token, c.ativada_em, c.desativada_em, c.microarea_id, c.ultimo_pacote_em, c.perfil_contato, c.perfil_eletrodependente, c0.id, c0.casa_token, c0.origem, c0.quando, c0.resultado
FROM motor.casa_monitorada AS c
LEFT JOIN motor.confirmacao AS c0 ON c.casa_token = c0.casa_token
WHERE c.microarea_id = @microarea AND c.desativada_em IS NULL
ORDER BY c.casa_token
   plano| Sort
   plano|   Sort Key: c.casa_token
   plano|   ->  Hash Right Join
   plano|         Hash Cond: (c0.casa_token = c.casa_token)
   plano|         ->  Seq Scan on confirmacao c0
   plano|         ->  Hash
   plano|               ->  Index Scan using ix_casa_monitorada_microarea_ativa on casa_monitorada c
   plano|                     Index Cond: (microarea_id = 3)

-- radar e entrada da avaliacao: eventos de 24 h mais as leituras baixas guardadas
-- @microarea='3'

   plano| Gather
   plano|   Workers Planned: 2
   plano|   ->  Hash Join

   plano|         ->  Parallel Seq Scan on evento_sensor e
   plano|               Filter: ((ocorrido_em >= '2026-11-17 13:31:00+00'::timestamp with time zone) OR (tensao_v < '100'::numeric))
   plano|         ->  Hash
   plano|               ->  Index Scan using ix_casa_monitorada_microarea_ativa on casa_monitorada c
